Query used 
WITH days AS (
      SELECT generate_series((CURRENT_DATE - INTERVAL '90 days')::date, CURRENT_DATE::date, '1 day')::date AS day
    ),
    daily AS (
      SELECT accounted_at::date AS day, sum(billable_loc) AS loc
      FROM loc_usage_ledger
      WHERE org_id = 151 AND status = 'accounted'
        AND accounted_at >= CURRENT_DATE - INTERVAL '90 days'
      GROUP BY 1
    )
    SELECT d.day, COALESCE(daily.loc, 0) AS loc
    FROM days d LEFT JOIN daily ON daily.day = d.day
    ORDER BY d.day;
