SELECT date_trunc('week', COALESCE(completed_at, created_at))::date AS week,
           trigger_type, count(*) AS n
    FROM reviews
    WHERE org_id = 151
      AND COALESCE(completed_at, created_at) >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY 1, 2
    ORDER BY 1;
