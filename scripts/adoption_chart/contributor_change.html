SELECT r.author_username,
           CASE WHEN l.accounted_at >= CURRENT_DATE - INTERVAL '45 days' THEN 'current' ELSE 'previous' END AS period,
           sum(l.billable_loc) AS loc
    FROM loc_usage_ledger l JOIN reviews r ON r.id = l.review_id
    WHERE l.org_id = 151 AND l.status = 'accounted' AND r.repository = 'LiveReview'
      AND r.author_username IS NOT NULL
      AND l.accounted_at >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY 1, 2;
