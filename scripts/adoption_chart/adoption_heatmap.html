Query used 
WITH days AS (
      SELECT generate_series(
        (CURRENT_DATE - INTERVAL '365 days')::date,
        CURRENT_DATE::date,
        '1 day'
      )::date AS day
    ),
    
        daily AS (
          SELECT date_trunc('day', COALESCE(completed_at, created_at))::date AS day, count(*) AS value
          FROM reviews
          WHERE org_id = 151
            AND COALESCE(completed_at, created_at) >= CURRENT_DATE - INTERVAL '365 days'
          GROUP BY 1
        )
        
    SELECT d.day, COALESCE(daily.value, 0) AS value
    FROM days d
    LEFT JOIN daily ON daily.day = d.day
    ORDER BY d.day;
