SELECT r.author_username AS engineer, f.vote_type, count(*) AS n
    FROM review_feedback f
    JOIN reviews r ON r.id = f.review_id
    WHERE f.org_id = 151 AND r.author_username IS NOT NULL
      AND f.retracted_at IS NULL
      AND f.created_at >= CURRENT_DATE - INTERVAL '365 days'
    GROUP BY 1, 2;
