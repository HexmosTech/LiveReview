Average 810.0 LOC per review across 440 reviews / 356394 LOC total.
Query used 
WITH days AS (
      SELECT generate_series((CURRENT_DATE - INTERVAL '90 days')::date, CURRENT_DATE::date, '1 day')::date AS day
    ),
    reviews_d AS (
      SELECT date_trunc('day', COALESCE(completed_at, created_at))::date AS day, count(*) AS n
      FROM reviews WHERE org_id = 151
        AND COALESCE(completed_at, created_at) >= CURRENT_DATE - INTERVAL '90 days'
      GROUP BY 1
    ),
    loc_d AS (
      SELECT accounted_at::date AS day, sum(billable_loc) AS loc
      FROM loc_usage_ledger WHERE org_id = 151 AND status = 'accounted'
        AND accounted_at >= CURRENT_DATE - INTERVAL '90 days'
      GROUP BY 1
    )
    SELECT d.day, COALESCE(reviews_d.n, 0) AS reviews, COALESCE(loc_d.loc, 0) AS loc
    FROM days d
    LEFT JOIN reviews_d ON reviews_d.day = d.day
    LEFT JOIN loc_d ON loc_d.day = d.day
    ORDER BY d.day;
